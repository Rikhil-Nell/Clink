import pandas as pd
import numpy as np
import json
import sys
from pyodide.http import pyfetch
from io import StringIO
import asyncio

# --- CONFIGURATION & ASSUMPTIONS ---
CONFIG = {
    "min_pair_support": 5,  # minimum times a pair must co-occur
    "min_item_support": 20,  # minimum invoices an item must appear in
    "top_n_items": 30,  # focus on top N most popular items
    "high_value_percentile": 0.8,  # top 20% of orders by value
    "min_lift_threshold": 1.1,  # minimum lift to consider significant
    "peak_hours_threshold": 0.1  # minimum 10% of orders to be peak hour
}

async def load_csv_from_url(url):
    """Load CSV from HTTP server using pyfetch"""
    response = await pyfetch(url)
    if not response.ok:
        raise Exception(f"HTTP {response.status}: {response.status_text}")
    csv_text = await response.string()
    csv_data = StringIO(csv_text)
    df = pd.read_csv(csv_data)
    return df

async def analyze_order_kpis(base_url):
    """Main analysis function for order-facing KPIs"""
    try:
        # Load all three CSV files
        invoice_df = await load_csv_from_url(f"{base_url}/order_analysis/Invoice_Aggregation.csv")
        lift_df = await load_csv_from_url(f"{base_url}/order_analysis/Item_Pair_Lift_Analysis.csv")
        cooccur_df = await load_csv_from_url(f"{base_url}/order_analysis/Product_Co_occurrence_Top30.csv")
        
    except Exception as e:
        return json.dumps({"error": f"Failed to load data: {str(e)}"}, indent=2)

    # Convert date columns if they exist
    if 'order_date' in invoice_df.columns:
        invoice_df['order_date'] = pd.to_datetime(invoice_df['order_date'])
    if 'order_date_only' in invoice_df.columns:
        invoice_df['order_date_only'] = pd.to_datetime(invoice_df['order_date_only'])

    # --- INVOICE ANALYSIS ---
    total_invoices = len(invoice_df)
    total_revenue = invoice_df['net_invoice_value'].sum()
    total_items_sold = invoice_df['total_quantity'].sum()
    
    # Order value distribution
    order_value_stats = invoice_df['net_invoice_value'].describe()
    high_value_threshold = invoice_df['net_invoice_value'].quantile(CONFIG['high_value_percentile'])
    high_value_orders = len(invoice_df[invoice_df['net_invoice_value'] >= high_value_threshold])
    
    # Basket size analysis
    basket_stats = invoice_df['total_items'].describe()
    
    # Day-of-week analysis
    if 'order_day' in invoice_df.columns:
        day_distribution = invoice_df['order_day'].value_counts().to_dict()
        peak_days = invoice_df['order_day'].value_counts().head(3).index.tolist()
    else:
        day_distribution = {}
        peak_days = []
    
    # Hour analysis if available
    hour_analysis = {}
    if 'order_date' in invoice_df.columns:
        invoice_df['hour'] = invoice_df['order_date'].dt.hour
        hourly_dist = invoice_df['hour'].value_counts().sort_index()
        total_orders_for_hour = len(invoice_df)
        peak_hours = hourly_dist[hourly_dist >= (total_orders_for_hour * CONFIG['peak_hours_threshold'])].index.tolist()
        hour_analysis = {
            "peak_hours": peak_hours,
            "hourly_distribution": hourly_dist.to_dict()
        }

    invoice_analysis = {
        "total_orders": total_invoices,
        "total_revenue": round(total_revenue, 2),
        "total_items_sold": int(total_items_sold),
        "average_order_value": round(total_revenue / total_invoices, 2) if total_invoices > 0 else 0,
        "average_items_per_order": round(total_items_sold / total_invoices, 2) if total_invoices > 0 else 0,
        "order_value_distribution": {
            "mean": round(order_value_stats['mean'], 2),
            "std": round(order_value_stats['std'], 2),
            "min": round(order_value_stats['min'], 2),
            "25%": round(order_value_stats['25%'], 2),
            "50%": round(order_value_stats['50%'], 2),
            "75%": round(order_value_stats['75%'], 2),
            "max": round(order_value_stats['max'], 2)
        },
        "high_value_orders": {
            "count": high_value_orders,
            "percentage": round(high_value_orders / total_invoices * 100, 2),
            "threshold": round(high_value_threshold, 2)
        },
        "basket_size_analysis": {
            "avg_unique_items": round(basket_stats['mean'], 2),
            "min_items": int(basket_stats['min']),
            "max_items": int(basket_stats['max']),
            "most_common_basket_size": int(basket_stats['50%'])
        },
        "temporal_patterns": {
            "day_of_week_distribution": day_distribution,
            "peak_days": peak_days,
            "hour_analysis": hour_analysis
        }
    }

    # --- LIFT ANALYSIS ---
    # Filter lift data based on config
    significant_lift = lift_df[
        (lift_df['lift'] >= CONFIG['min_lift_threshold']) & 
        (lift_df['count'] >= CONFIG['min_pair_support'])
    ].copy()
    
    top_lift_pairs = significant_lift.head(20) if len(significant_lift) > 0 else pd.DataFrame()
    
    lift_analysis = {
        "total_item_pairs_analyzed": len(lift_df),
        "significant_pairs_found": len(significant_lift),
        "min_lift_threshold": CONFIG['min_lift_threshold'],
        "min_support_threshold": CONFIG['min_pair_support'],
        "top_lift_pairs": []
    }
    
    if len(top_lift_pairs) > 0:
        for _, row in top_lift_pairs.iterrows():
            lift_analysis["top_lift_pairs"].append({
                "item_1": row['item_1'],
                "item_2": row['item_2'],
                "lift": round(row['lift'], 3),
                "co_occurrence_count": int(row['count']),
                "interpretation": f"Customers who buy {row['item_1']} are {row['lift']:.1f}x more likely to also buy {row['item_2']}"
            })
    
    # --- CO-OCCURRENCE ANALYSIS ---
    # Since we're working with the top 30 filtered matrix, convert it for analysis
    cooccur_items = list(cooccur_df.index) if hasattr(cooccur_df, 'index') else []
    
    # If it's a matrix, find the strongest co-occurrences
    strongest_cooccurrences = []
    if len(cooccur_items) > 0:
        # Convert matrix to pairs format for easier analysis
        for i, item1 in enumerate(cooccur_items):
            for j, item2 in enumerate(cooccur_items):
                if i < j:  # Avoid duplicates
                    try:
                        count = cooccur_df.iloc[i, j] if hasattr(cooccur_df, 'iloc') else 0
                        if count > 0:
                            strongest_cooccurrences.append({
                                "item_1": item1,
                                "item_2": item2,
                                "count": int(count)
                            })
                    except:
                        continue
    
    # Sort by count and take top pairs
    strongest_cooccurrences = sorted(strongest_cooccurrences, key=lambda x: x['count'], reverse=True)[:15]
    
    cooccurrence_analysis = {
        "analysis_scope": f"Top {CONFIG['top_n_items']} most popular items",
        "items_in_matrix": len(cooccur_items),
        "strongest_cooccurrences": strongest_cooccurrences,
        "matrix_summary": {
            "total_possible_pairs": len(cooccur_items) * (len(cooccur_items) - 1) // 2,
            "pairs_with_cooccurrence": len(strongest_cooccurrences)
        }
    }

    # --- BUSINESS INSIGHTS ---
    business_insights = {
        "bundle_opportunities": [],
        "cross_sell_recommendations": [],
        "inventory_insights": []
    }
    
    # Generate bundle opportunities from high lift pairs
    if len(top_lift_pairs) > 0:
        for _, row in top_lift_pairs.head(5).iterrows():
            business_insights["bundle_opportunities"].append({
                "bundle_items": [row['item_1'], row['item_2']],
                "statistical_strength": f"Lift: {row['lift']:.2f}",
                "frequency": f"Appears together in {row['count']} orders",
                "recommendation": f"Strong candidate for bundle pricing or combo offers"
            })
    
    # Cross-sell from co-occurrence
    if len(strongest_cooccurrences) > 0:
        for pair in strongest_cooccurrences[:5]:
            business_insights["cross_sell_recommendations"].append({
                "trigger_item": pair['item_1'],
                "suggest_item": pair['item_2'],
                "frequency": f"Bought together {pair['count']} times",
                "strategy": "Suggest as add-on during ordering"
            })

    # --- FINAL SUMMARY ---
    final_summary = {
        "data_sources": {
            "invoice_data": f"{base_url}/Invoice_Aggregation.csv",
            "lift_analysis": f"{base_url}/Item_Pair_Lift_Analysis.csv",
            "cooccurrence_matrix": f"{base_url}/Product_Co_occurrence_Top30.csv"
        },
        "analysis_timestamp": pd.Timestamp.now().isoformat(),
        "analysis_config": CONFIG,
        "invoice_analysis": invoice_analysis,
        "lift_analysis": lift_analysis,
        "cooccurrence_analysis": cooccurrence_analysis,
        "business_insights": business_insights
    }

    return json.dumps(final_summary, indent=2)

async def main():
    """Main execution function"""
    base_url = "http://localhost:8000"
    
    if len(sys.argv) == 2:
        base_url = sys.argv[1]
    
    summary_json = await analyze_order_kpis(base_url)
    print(summary_json)

# Execute the analysis
loop = asyncio.get_event_loop()
loop.run_until_complete(main())
